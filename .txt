gcc -o game.exe main.c include/menu.c -I. -Istb-master -lopengl32 -lglu32 -lgdi32 -std=c99 -Wall

gcc - компилятор
-o game.exe - выходной файл
main.c - главный файл
include/menu.c - файл с меню
-I. - собрать файлы из текущей папки
-Istb-master - собрать файлы из stb-master
-lopengl32 -lglu32 -lgdi32 - подключение нужных библиотек
-std=c99 - правила сборки проекта
-Wall
